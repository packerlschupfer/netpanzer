[binaries]
c = 'aarch64-linux-gnu-gcc'
cpp = 'aarch64-linux-gnu-g++'
ar = 'aarch64-linux-gnu-ar'
strip = 'aarch64-linux-gnu-strip'
ld = 'aarch64-linux-gnu-ld'
# Without this meson has no pkg-config for the host machine and any
# dependency() lookup fails outright -- which is how the SDL3 wrap, which
# requires udev/dbus/wayland from the system, could not configure at all.
# Both spellings: meson took 'pkgconfig' historically and 'pkg-config' now.
pkgconfig = 'aarch64-linux-gnu-pkg-config'
pkg-config = 'aarch64-linux-gnu-pkg-config'

[host_machine]
system = 'linux'
# Meson autodetects the subsystem for a native build but cannot for a cross
# one, and SDL3's hidapi meson checks host_machine.subsystem() -- without this
# the whole configure dies with "Subsystem not defined or could not be
# autodetected".
subsystem = 'linux'
cpu_family = 'aarch64'
cpu = 'aarch64'
endian = 'little'
